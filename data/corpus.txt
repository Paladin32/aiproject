# Python examples
def add(a, b):
    return a + b

def subtract(a, b):
    return a - b

def multiply(a, b):
    return a * b

def factorial(n):
    if n <= 1:
        return 1
    return n * factorial(n - 1)

def fibonacci(n):
    sequence = [0, 1]
    for index in range(2, n):
        sequence.append(sequence[-1] + sequence[-2])
    return sequence[:n]

def is_even(number):
    return number % 2 == 0

def greet(name):
    return f"Hello, {name}!"

class Calculator:
    def add(self, left, right):
        return left + right

    def divide(self, left, right):
        if right == 0:
            raise ValueError("Cannot divide by zero")
        return left / right
